// A local, event-driven menu. Repository choices only change display labels.
function repoCompanionInstallMenu(dispatch) {
  let menu, previousFocus;
  function close(restore = false) {
    if (!menu) return;
    menu.remove(); menu = null;
    if (restore && previousFocus?.isConnected) previousFocus.focus({preventScroll:true});
    previousFocus = null;
  }
  function open(event) {
    const target = event.target?.closest?.(`[data-vscode-context]`);
    if (!target) return;
    let context;
    try { context = JSON.parse(target.getAttribute(`data-vscode-context`)); } catch { return; }
    const key = context?.repoCompanionConversationKey;
    if (context?.webviewSection !== `repoCompanionChat` || !/^local\/[0-9a-f]{8}(?:-[0-9a-f]{4}){3}-[0-9a-f]{12}$/i.test(key)) return;
    const repos = window.__codexRepoCompanionRepositories;
    if (!Array.isArray(repos)) return; // Older bridges retain the native picker menu.
    const customOnly = repos.length <= 1;
    event.preventDefault(); event.stopImmediatePropagation(); close();
    previousFocus = document.activeElement;
    menu = document.createElement(`div`);
    menu.id = `repo-companion-menu`;
    menu.setAttribute(`role`, `dialog`);
    menu.setAttribute(`aria-label`, customOnly ? `Custom chat label` : `Choose chat repository`);
    Object.assign(menu.style, {position:`fixed`,zIndex:`2147483647`,width:`min(380px, calc(100vw - 16px))`,maxHeight:`calc(100vh - 16px)`,display:`flex`,flexDirection:`column`,boxSizing:`border-box`,padding:`6px`,border:`1px solid var(--vscode-menu-border, #555)`,borderRadius:`6px`,background:`var(--vscode-menu-background, #252526)`,color:`var(--vscode-menu-foreground, #eee)`,boxShadow:`0 4px 16px #0006`,font:`13px var(--vscode-font-family, sans-serif)`});
    const heading = document.createElement(`div`);
    heading.textContent = window.__codexRepoCompanionPinManualLabels === false ? `Choose repository (follows discussion)` : `Choose repository (pins label)`;
    if (customOnly) heading.textContent = window.__codexRepoCompanionPinManualLabels === false ? `Custom label (follows discussion)` : `Custom label (pins label)`;
    Object.assign(heading.style, {padding:`5px 8px`,opacity:`0.75`});
    const search = document.createElement(`input`);
    search.type = `search`; search.placeholder = `Filter repositories`;
    search.setAttribute(`aria-label`, `Filter repositories`);
    if (customOnly) {
      search.type = `text`; search.placeholder = `Enter a custom label`; search.maxLength = 100;
      search.setAttribute(`aria-label`, `Custom label`);
      search.value = window.__codexRepoCompanionCustomLabels?.[key] || ``;
    }
    Object.assign(search.style, {margin:`4px`,padding:`5px`,minWidth:`0`,color:`var(--vscode-input-foreground, #eee)`,background:`var(--vscode-input-background, #333)`,border:`1px solid var(--vscode-input-border, #666)`});
    const list = document.createElement(`div`);
    list.setAttribute(`role`, `group`); list.setAttribute(`aria-label`, `Repositories`);
    Object.assign(list.style, {overflowY:`auto`,minHeight:`0`,maxHeight:`55vh`,flex:`1 1 auto`});
    function choose(action, root, label) {
      close(true);
      dispatch(`repo-companion-menu-action`, {key,action,...(root ? {root} : {}),...(label !== undefined ? {label} : {})});
    }
    function button(label, action, root, title) {
      const item = document.createElement(`button`);
      item.type = `button`; item.textContent = label; item.title = title || label;
      item.dataset.action = action;
      if (root) item.dataset.root = root;
      Object.assign(item.style, {display:`block`,width:`100%`,textAlign:`left`,padding:`6px 8px`,border:`0`,borderRadius:`3px`,background:`transparent`,color:`inherit`,font:`inherit`,cursor:`pointer`,whiteSpace:`nowrap`,overflow:`hidden`,textOverflow:`ellipsis`,flexShrink:`0`});
      item.onfocus = item.onmouseenter = () => { item.style.background = `var(--vscode-menu-selectionBackground, #094771)`; item.style.color = `var(--vscode-menu-selectionForeground, #fff)`; };
      item.onblur = item.onmouseleave = () => { item.style.background = `transparent`; item.style.color = `inherit`; };
      item.onclick = event => { event.preventDefault(); event.stopPropagation(); choose(action,root); };
      return item;
    }
    function populate() {
      list.replaceChildren();
      const query = search.value.toLocaleLowerCase();
      const labels = new Map();
      for (const repo of repos) labels.set(repo.label, (labels.get(repo.label) || 0) + 1);
      for (const repo of repos) {
        if (!(repo.label + ` ` + repo.description).toLocaleLowerCase().includes(query)) continue;
        const label = labels.get(repo.label) > 1 ? repo.label + ` - ` + repo.description : repo.label;
        list.append(button(label, `assign`, repo.root, repo.description));
      }
      if (!list.childElementCount) {
        const empty = document.createElement(`div`);
        empty.textContent = repos.length ? `No matching repositories` : `No Git repositories open in this workspace`;
        empty.style.padding = `8px`; list.append(empty);
      }
    }
    if (!customOnly) { search.oninput = populate; populate(); }
    const separator = document.createElement(`hr`);
    Object.assign(separator.style,{width:`100%`,border:`0`,borderTop:`1px solid var(--vscode-menu-separatorBackground, #555)`,margin:`5px 0`});
    menu.append(heading,search);
    if (customOnly) {
      const apply = button(`Apply Label`,`applyCustom`);
      const error = document.createElement(`div`);
      error.setAttribute(`role`,`alert`); error.style.padding = `4px 8px`;
      error.style.color = `var(--vscode-errorForeground, #f88)`;
      apply.onclick = event => {
        event.preventDefault(); event.stopPropagation();
        if (!search.value.trim() || search.value.trim().length > 100 || /[\x00-\x1f\x7f\[\]]/.test(search.value)) {
          error.textContent = `Enter 1-100 characters without brackets or control characters.`; search.focus(); return;
        }
        choose(`custom`,undefined,search.value.trim());
      };
      menu.append(apply,error);
    } else { menu.append(list,button(`Custom Label...`,`custom`)); }
    menu.append(separator,button(`Chat Colour...`,`colour`),button(`Repository Colour...`,`repositoryColour`),button(window.__codexRepoCompanionStarred?.includes(key)?`Unstar Chat`:`Star Chat`,`star`),button(`Starred Chats...`,`starred`),button(`Use Automatic Labels`,`auto`),button(`Clear Labels`,`clear`),button(`New Chat in Sidebar`,`newChat`));
    menu.addEventListener(`keydown`,event=>{
      if (event.key === `Escape`) { event.preventDefault(); event.stopPropagation(); close(true); return; }
      if (event.key === `Tab`) { close(true); return; }
      const items = Array.from(menu.querySelectorAll(`button`));
      if (event.key === `Enter` && document.activeElement === search) { event.preventDefault(); items.find(item=>customOnly ? item.dataset.action === `applyCustom` : item.dataset.root)?.click(); return; }
      if (![`ArrowDown`,`ArrowUp`,`Home`,`End`].includes(event.key)) return;
      if (document.activeElement === search && [`Home`,`End`].includes(event.key)) return;
      event.preventDefault(); event.stopPropagation();
      const index = items.indexOf(document.activeElement);
      const next = event.key === `Home` ? 0 : event.key === `End` ? items.length - 1 : event.key === `ArrowDown` ? (index + 1) % items.length : (index <= 0 ? items.length - 1 : index - 1);
      items[next]?.focus();
    });
    document.body.append(menu);
    const bounds = menu.getBoundingClientRect(), anchor = target.getBoundingClientRect();
    menu.style.left = Math.max(8,Math.min(event.clientX || anchor.left,window.innerWidth-bounds.width-8)) + `px`;
    menu.style.top = Math.max(8,Math.min(event.clientY || anchor.bottom,window.innerHeight-bounds.height-8)) + `px`;
    search.focus({preventScroll:true});
  }
  const outside = event => { if (menu && !menu.contains(event.target)) close(); };
  const blur = () => close();
  window.addEventListener(`contextmenu`,open,true);
  window.addEventListener(`pointerdown`,outside,true);
  window.addEventListener(`blur`,blur);
  return () => { close(); window.removeEventListener(`contextmenu`,open,true); window.removeEventListener(`pointerdown`,outside,true); window.removeEventListener(`blur`,blur); };
}
