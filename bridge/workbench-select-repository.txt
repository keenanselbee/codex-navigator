xt.registerCommandAndKeybindingRule({id:"codexRepoCompanion.workbench.selectRepository",weight:200,handler:(accessor,request)=>{
if(!request||typeof request.root!=="string"||!request.root.startsWith("file:")||typeof request.providerId!=="string"||!/^scm\d+$/.test(request.providerId))throw new Error("Invalid repository selection request.");
const service=accessor.get(lc),view=accessor.get(Au);
const matches=Array.from(service.repositories).filter(repo=>repo.provider.id===request.providerId&&repo.provider.rootUri?.scheme==="file"&&repo.provider.rootUri.toString().toLowerCase()===request.root.toLowerCase());
if(matches.length!==1)throw new Error("The exact Git repository is not registered in this window.");
const target=matches[0],before=view.visibleRepositories;
if(!before.includes(target)){
if(view.selectionModeConfig.get()!=="multiple")throw new Error("Make the assigned repository visible in single-selection mode first.");
view.visibleRepositories=[...before,target];
}
try{view.focus(target);if(view.focusedRepository!==target)throw new Error("Direct repository selection was not confirmed.");}
catch(error){view.visibleRepositories=before;throw error;}
return true;
}});
